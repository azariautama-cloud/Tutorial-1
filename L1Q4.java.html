import javax.swing.*;
import java.awt.*;

public class L1Q4 extends JPanel {

    public void paintComponent(Graphics g) {
        super.paintComponent(g);

        String[] month = {
                "January", "February", "March",
                "April", "May", "June"
        };

        int[] amount = {2500, 1600, 2000, 2700, 3200, 800};

        g.drawString("Total Sales of Product A", 280, 40);

        g.drawLine(50, 350, 700, 350);
        g.drawLine(50, 70, 50, 350);

        for (int i = 0; i < 6; i++) {

            int x = 80 + (i * 100);
            int height = amount[i] / 12;

            g.setColor(Color.BLUE);
            g.fillRect(x, 350 - height, 50, height);

            g.setColor(Color.BLACK);
            g.drawString("" + amount[i], x, 340 - height);
            g.drawString(month[i], x - 10, 370);
        }
    }

    public static void main(String[] args) {

        JFrame frame = new JFrame("Sales Chart");

        frame.add(new L1Q4());
        frame.setSize(750, 450);
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setVisible(true);
    }
}
